% 1. Typologie des requêtes
\section*{1. Typologie des Requêtes Utilisateurs (Intents)}
\begin{itemize}
    \item \textbf{\texttt{check\_availability}} : Recherche de stock ou d'officine disponible.\newline
    \textit{Utterances brutes :} \texttt{"khassni doliprane 1g 3afak fin nqder nlqah daba casablanca"} / \texttt{"dispo doli sirop enfant ?"}
    \item \textbf{\texttt{ask\_price\_reimbursement}} : Demande du Prix Public de Vente (PPV) et remboursement CNOPS/AMO.\newline
    \textit{Utterances brutes :} \texttt{"chhal thaman dial augmantin 1g wa wach remboursable cnops"} / \texttt{"prix du kardegic 75"}
    \item \textbf{\texttt{find\_alternatives}} : Recherche de génériques ou de substitution DCI.\newline
    \textit{Utterances brutes :} \texttt{"maqitch amoxil achnou nqder nakhd f blastou"} / \texttt{"remplaçant de clamoxyl 1g"}
    \item \textbf{\texttt{ask\_posology\_usage}} : Informations d'usage et posologie usuelle.\newline
    \textit{Utterances brutes :} \texttt{"chhal mn habba f nhar dial spafon"} / \texttt{"doliprane 500 avant ou après le repas ?"}
    \item \textbf{\texttt{ask\_side\_effects\_safety}} : Contre-indications et alertes de sécurité.\newline
    \textit{Utterances brutes :} \texttt{"wach doliprane khayf l la femme enceinte"} / \texttt{"effets secondaires du daflon"}
\end{itemize}

% 2. Pipeline NLP & Matching
\section*{2. Pipeline NLP \& Extraction d'Entités (NER \& Matching)}
\begin{itemize}
    \item \textbf{Normalisation \& Trilinguisme (Darija / Arabizi / FR / Arabe) :}
    Translittération Arabizi $\rightarrow$ FR/Arabe via dictionnaire phonétique (mappage \texttt{'3'}$\rightarrow$Ain, \texttt{'7'}$\rightarrow$Haa, \texttt{'9'}$\rightarrow$Qaf). Tokenisation subword (CamemBERT / DarijaBERT). Suppression des diacritiques, normalisation Unicode NDFC et passage en minuscules.
    \item \textbf{Résolution Floue \& Phonétique des Marques :}
    Recherche hybride 2-Tier sur le référentiel CNOPS/DMP (5 918 SKUs) :
    \begin{enumerate}[label=\alph*),itemsep=1pt,topsep=1pt]
        \item \textit{Tier 1 (Lexical/Phonétique)} : \textbf{SymSpell} ($d \le 2$) couplé à un algorithme \textbf{Double Metaphone} adapté au latin/français pour absorber les fautes de frappe (\textit{"augmantain"} $\rightarrow$ \texttt{AUGMENTIN}, \textit{"doli"} $\rightarrow$ \texttt{DOLIPRANE}).
        \item \textit{Tier 2 (Sémantique)} : Bi-encoder dense (\texttt{Sentence-BERT}) calculant la similarité cosinus (seuil $> 0.82$).
    \end{enumerate}
    \item \textbf{Désambiguïsation Multi-Attributs :} Modèle Spacy NER entraîné à extraire conjointement les slots \texttt{[BRAND\_NAME]}, \texttt{[DCI]}, \texttt{[DOSAGE\_VAL]}, \texttt{[DOSAGE\_UNIT]}, \texttt{[GALENIC\_FORM]}, \texttt{[PACKAGING]}.\newline
    \textit{Exemple :} \texttt{"doliprane 1000 mg gélule bte de 8"} $\rightarrow$ \texttt{\{DCI: "PARACETAMOL", Brand: "DOLIPRANE", Dosage: 1000, Form: "GELULE"\}}.
\end{itemize}

% 3. Garde-fous & Fallbacks
\section*{3. Gestion des Cas Limites \& Garde-fous (Safety \& Fallbacks)}
\begin{itemize}
    \item \textbf{Seuils de Confiance NLU :} 
    $\text{Score} \ge 0.85$ : Exécution directe API Métier. $|$ $0.60 \le \text{Score} < 0.85$ : \textbf{Confirmation Active} via boutons d'action UI (sélection parmi top-3 molécules/dosages). $|$ $\text{Score} < 0.60$ : Fallback vers reformulation guidée.
    \item \textbf{Garde-fous Médicaux \& Sécurité :} Détection automatique d'expressions d'urgence ou de diagnostic médical (\textit{"j'ai une douleur thoracique"}) $\rightarrow$ Interruption immédiate du flux conversationnel, affichage du numéro d'urgence (150/141) et redirection officine.
    \item \textbf{Disclaimer Légal Obligatoire :} Mentions légales systématiques : \textit{"Assistant d'information logistique et tarifaire. Ne remplace pas la prescription d'un médecin ni le conseil du pharmacien."}
\end{itemize}
